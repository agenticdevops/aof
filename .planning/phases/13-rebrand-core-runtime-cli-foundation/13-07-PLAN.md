    **cli.rs — New command structure:**

    ```rust
    #[derive(Parser)]
    #[command(name = "agentix", version, about = "OpenAgentiX — Enterprise Agent Automation Platform")]
    pub struct Cli {
        /// Gateway URL (default: http://127.0.0.1:7777)
        #[arg(long, global = true, env = "AGENTIX_GATEWAY_URL", default_value = "http://127.0.0.1:7777")]
        pub gateway_url: String,

        /// Output format: text, json
        #[arg(long, global = true, default_value = "text")]
        pub output: OutputFormat,

        /// Quiet mode — final result only, no ReAct loop details
        #[arg(long, short, global = true)]
        pub quiet: bool,

        #[command(subcommand)]
        pub command: Commands,
    }

    #[derive(Subcommand)]
    pub enum Commands {
        /// Start or manage the gateway server
        Gateway {
            #[command(subcommand)]
            command: GatewayCommands,
        },

        /// List agents registered in the gateway
        Agents {
            /// Filter by namespace
            #[arg(long, short)]
            namespace: Option<String>,
        },

        /// Show run history
        Runs {
            /// Agent name (optional — shows all agents if omitted)
            agent: Option<String>,

            /// Show only last N runs
            #[arg(long, default_value = "20")]
            limit: usize,
        },

        /// Show agent logs
        Logs {
            /// Agent name
            agent: String,

            /// Specific run ID (latest if omitted)
            #[arg(long)]
            run: Option<String>,

            /// Follow log output in real-time
            #[arg(long, short)]
            follow: bool,
        },

        /// Stop a running agent
        Stop {
            /// Agent name
            agent: String,

            /// Run ID to stop (latest running if omitted)
            #[arg(long)]
            run: Option<String>,
        },

        /// Apply agent configuration to the gateway
        Apply {
            /// YAML file path
            #[arg(short, long)]
            file: String,
        },

        /// Validate agent YAML locally (no gateway needed)
        Validate {
            /// YAML file path
            file: String,
        },

        /// Show version information
        Version,
    }

    #[derive(Subcommand)]
    pub enum GatewayCommands {
        /// Start the gateway server
        Start {
            /// Workspace config file (default: ./agentix.yaml)
            #[arg(long, short, default_value = "./agentix.yaml")]
            config: String,

            /// Override agents directory
            #[arg(long)]
            agents_dir: Option<String>,

            /// Override port
            #[arg(long, short)]
            port: Option<u16>,

            /// Override host
            #[arg(long)]
            host: Option<String>,
        },
        /// Show gateway status
        Status,
    }

    #[derive(Clone, ValueEnum)]
    pub enum OutputFormat { Text, Json }
    ```

    **client.rs — Gateway HTTP client:**

    ```rust
    pub struct GatewayClient {
        base_url: String,
        client: reqwest::Client,
    }

    impl GatewayClient {
        pub fn new(base_url: &str) -> Self { ... }

        pub async fn list_agents(&self) -> Result<Vec<AgentSummary>> { GET /api/v1/agents }
        pub async fn get_agent(&self, name: &str) -> Result<AgentDetail> { GET /api/v1/agents/:name }
        pub async fn run_agent(&self, name: &str, input: &str) -> Result<impl Stream<Item = ReActEvent>> { POST /api/v1/agents/:name/run with SSE parsing }
        pub async fn list_runs(&self, agent: Option<&str>, limit: usize) -> Result<Vec<RunSummary>> { GET /api/v1/agents/:name/runs }
        pub async fn get_run_logs(&self, agent: &str, run_id: &str) -> Result<Vec<ReActEvent>> { GET /api/v1/agents/:name/runs/:id/logs }
        pub async fn stop_run(&self, agent: &str, run_id: &str) -> Result<()> { DELETE /api/v1/agents/:name/runs/:id }
        pub async fn register_agent(&self, yaml: &str) -> Result<AgentApplyResponse> { POST /api/v1/agents with {"yaml": yaml} }
        pub async fn update_agent(&self, name: &str, yaml: &str) -> Result<AgentApplyResponse> { PUT /api/v1/agents/:name with {"yaml": yaml} }
        pub async fn health(&self) -> Result<bool> { GET /healthz }
    }
    ```
