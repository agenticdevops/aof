    **cli.rs — Complete command structure:**

    ```rust
    #[derive(Parser)]
    #[command(
        name = "agentix",
        version,
        about = "OpenAgentiX — Enterprise Agent Automation Platform",
        long_about = "Define AI agents as directories. Run them on demand. Track everything.\nhttps://openagentix.org"
    )]
    pub struct Cli {
        /// Gateway URL
        #[arg(long, global = true, env = "AGENTIX_GATEWAY_URL", default_value = "http://127.0.0.1:7777")]
        pub gateway_url: String,

        /// Output format: text or json
        #[arg(long, global = true, default_value = "text", value_enum)]
        pub output: OutputFormat,

        /// Suppress ReAct loop details, show final result only
        #[arg(long, short, global = true)]
        pub quiet: bool,

        #[command(subcommand)]
        pub command: Commands,
    }

    #[derive(Subcommand)]
    pub enum Commands {
        /// Manage the gateway server
        Gateway {
            #[command(subcommand)]
            command: GatewayCommands,
        },
        /// List agents registered in the gateway
        Agents {
            #[arg(long, short)]
            namespace: Option<String>,
        },
        /// Show agent run history
        Runs {
            /// Agent name (all agents if omitted)
            agent: Option<String>,
            #[arg(long, default_value = "20")]
            limit: usize,
        },
        /// Show agent execution logs
        Logs {
            agent: String,
            #[arg(long)]
            run: Option<String>,
            #[arg(long, short)]
            follow: bool,
        },
        /// Stop a running agent
        Stop {
            agent: String,
            #[arg(long)]
            run: Option<String>,
        },
        /// Apply agent configuration to the gateway
        Apply {
            #[arg(short, long)]
            file: String,
        },
        /// Validate agent YAML or agent directory (no gateway needed)
        Validate {
            /// Path to agent.yaml file, flat YAML file, or agent directory
            path: String,
        },
        /// Print version information
        Version,
    }

    #[derive(Subcommand)]
    pub enum GatewayCommands {
        /// Start the gateway server
        Start {
            #[arg(long, short, default_value = "./agentix.yaml")]
            config: String,
            #[arg(long)]
            agents_dir: Option<String>,
            #[arg(long, short)]
            port: Option<u16>,
        },
        /// Show gateway status
        Status,
    }

    #[derive(Clone, ValueEnum)]
    pub enum OutputFormat { Text, Json }
    ```

    **main.rs:**
    ```rust
    #[tokio::main]
    async fn main() {
        let cli = Cli::parse();

        // Tracing setup — error level unless RUST_LOG set
        tracing_subscriber::fmt()
            .with_env_filter(tracing_subscriber::EnvFilter::from_default_env())
            .with_target(false)
            .init();

        // Color detection: disable if not TTY or NO_COLOR is set
        if !std::io::IsTerminal::is_terminal(&std::io::stdout()) || std::env::var("NO_COLOR").is_ok() {
            colored::control::set_override(false);
        }

        let result = match cli.command {
            Commands::Gateway { command } => commands::gateway::run(command).await,
            Commands::Agents { namespace } => commands::agents::run(&cli, namespace).await,
            Commands::Runs { agent, limit } => commands::runs::run(&cli, agent, limit).await,
            Commands::Logs { agent, run, follow } => commands::logs::run(&cli, agent, run, follow).await,
            Commands::Stop { agent, run } => commands::stop::run(&cli, agent, run).await,
            Commands::Apply { file } => commands::apply::run(&cli, file).await,
            Commands::Validate { path } => commands::validate::run(path).await,
            Commands::Version => commands::version::run(),
        };

        if let Err(e) = result {
            eprintln!("{}", if colored::control::SHOULD_COLORIZE.should_colorize() {
                format!("\x1b[31merror:\x1b[0m {}", e)
            } else {
                format!("error: {}", e)
            });
            std::process::exit(1);
        }
    }
    ```

    - `--quiet`: print only the final output text (extract from the `Complete` event)

    **stop.rs:**
    - If no --run, list runs and find the currently-running one for the agent
    - Call `client.stop_run(agent, run_id)`. Print "Stopped run {id} for agent {agent}".

    **apply.rs:**
    - Read file at path. Detect if path is a directory (agent directory format) or a flat YAML file.
    - For flat YAML: read content, validate locally with `AgentLoader::load_yaml_str(content)` (or parse + validate inline).
    - For agent directory: validate with `DirectoryLoader::load(path)` — no bytes sent to gateway (directory isn't uploadable). Instead: print "Agent directory validated locally. Use `agentix gateway start` with the agents_dir containing this agent to register it." Return early.
    - For flat YAML that validates successfully: send to gateway via `client.register_agent(yaml_content)`.

      To produce this: parse with `AgentLoader`, get the AgentixError, extract field name from error message, find matching line in the YAML text, render the pointer.
