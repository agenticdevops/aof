```rust
use async_trait::async_trait;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::fmt;
use std::sync::Arc;

use crate::AgentixError;

/// The trigger source identifies what fired this agent run.
#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum TriggerSource {
    Cron,
    Webhook,
    GitHub,
    Jira,
    Slack,
    Discord,
    Telegram,
    Agent,
    Cli,
}

impl fmt::Display for TriggerSource {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        let s = match self {
            Self::Cron => "cron",
            Self::Webhook => "webhook",
            Self::GitHub => "github",
            Self::Jira => "jira",
            Self::Slack => "slack",
            Self::Discord => "discord",
            Self::Telegram => "telegram",
            Self::Agent => "agent",
            Self::Cli => "cli",
        };
        write!(f, "{}", s)
    }
}

/// Envelope delivered to every triggered agent run.
/// The agent receives this via its input context.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TriggerEvent {
    /// What fired this trigger
    pub source: TriggerSource,
    /// Platform-specific payload (PR object, cron metadata, webhook body, etc.)
    pub payload: serde_json::Value,
    /// Arbitrary key/value context (repo, channel_id, agent_caller, etc.)
    #[serde(default)]
    pub context: HashMap<String, String>,
    /// When the event was created
    pub fired_at: chrono::DateTime<chrono::Utc>,
    /// The trigger_id that produced this event (matches TriggerRegistry key)
    pub trigger_id: String,
}

impl TriggerEvent {
    pub fn new(
        source: TriggerSource,
        payload: serde_json::Value,
        trigger_id: impl Into<String>,
    ) -> Self {
        Self {
            source,
            payload,
            context: HashMap::new(),
            fired_at: chrono::Utc::now(),
            trigger_id: trigger_id.into(),
        }
    }

    pub fn with_context(mut self, key: impl Into<String>, value: impl Into<String>) -> Self {
        self.context.insert(key.into(), value.into());
        self
    }
}

/// Pluggable trigger interface. New trigger types can be added without modifying core.
/// Implementations: CronTrigger, WebhookTrigger, GitHubTrigger, JiraTrigger,
///                  ChannelMentionTrigger, AgentTrigger, CliTrigger.
#[async_trait]
pub trait TriggerTrait: Send + Sync {
    /// Return the unique identifier of this trigger instance.
    fn trigger_id(&self) -> &str;

    /// Return the trigger source type.
    fn source(&self) -> TriggerSource;

    /// Start the trigger (for long-running listeners — cron loops, webhook servers, etc.).
    /// Should send TriggerEvent via the provided sender when fired.
    async fn start(
        &self,
        sender: tokio::sync::mpsc::Sender<(String, TriggerEvent)>,
    ) -> Result<(), AgentixError>;

    /// Gracefully stop the trigger.
    async fn stop(&self) -> Result<(), AgentixError>;
}

/// Holds all registered TriggerTrait objects keyed by trigger_id.
/// Owned by the gateway and started at boot.
#[derive(Default)]
pub struct TriggerRegistry {
    triggers: HashMap<String, Arc<dyn TriggerTrait>>,
}

impl TriggerRegistry {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn register(&mut self, trigger: Arc<dyn TriggerTrait>) {
        self.triggers.insert(trigger.trigger_id().to_string(), trigger);
    }

    pub fn get(&self, id: &str) -> Option<Arc<dyn TriggerTrait>> {
        self.triggers.get(id).cloned()
    }

    pub fn all(&self) -> Vec<Arc<dyn TriggerTrait>> {
        self.triggers.values().cloned().collect()
    }

    pub fn len(&self) -> usize {
        self.triggers.len()
    }

    pub fn is_empty(&self) -> bool {
        self.triggers.is_empty()
    }
}
```
