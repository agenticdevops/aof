**Step 2: Implement CoordinatorProtocol for AgentManager.**

```rust
#[async_trait]
impl CoordinatorProtocol for AgentManager {
    async fn delegate(
        &self,
        target: &str,
        task: &str,
        payload: serde_json::Value,
    ) -> AgentixResult<DelegationResult> {
        let delegation_id = uuid::Uuid::new_v4().to_string();

        // Build delegation message
        let msg = DelegationMessage {
            id: delegation_id.clone(),
            from_agent: "coordinator".to_string(), // caller sets via payload context
            task: task.to_string(),
            payload: payload.clone(),
            reply_to: None,
        };

        // Send to target agent's inbox (create a TriggerEvent wrapping it)
        let trigger_event = TriggerEvent::new(
            TriggerSource::Agent,
            serde_json::json!({
                "delegation_id": delegation_id,
                "task": task,
                "payload": payload,
            }),
            &delegation_id,
        ).with_context("delegation_id", &delegation_id);

        // Run the target agent
        let run_result = self.run_agent_with_trigger(target, trigger_event).await?;

        // Append audit event
        self.append_audit_event(AuditEvent {
            event_type: "delegation".to_string(),
            actor: "coordinator".to_string(),
            target: Some(target.to_string()),
            delegation_id: Some(delegation_id.clone()),
            timestamp: chrono::Utc::now(),
            outcome: if run_result.success { "success" } else { "failure" }.to_string(),
        }).await;

        Ok(DelegationResult {
            delegation_id,
            from_agent: target.to_string(),
            output: run_result.output,
            status: if run_result.success {
                DelegationStatus::Success
            } else {
                DelegationStatus::Failure
            },
            completed_at: chrono::Utc::now(),
        })
    }

    async fn delegate_parallel(
        &self,
        delegations: Vec<(String, String, serde_json::Value)>,
    ) -> AgentixResult<Vec<DelegationResult>> {
        let futures: Vec<_> = delegations
            .into_iter()
            .map(|(target, task, payload)| {
                let mgr = self; // Note: need Arc<Self> pattern for multi-task
                async move { mgr.delegate(&target, &task, payload).await }
            })
            .collect();

        // Use futures::future::join_all for parallel execution
        let results = futures::future::join_all(futures).await;
        results.into_iter().collect::<AgentixResult<Vec<_>>>()
    }
}
```

Note: Use `Arc<AgentManager>` if self-reference is needed for async tasks. Adjust as needed to pass borrow checker — use `Arc<Self>` and clone for each spawned future.

```

  "payload": {"host": "pg-primary", "threshold_ms": 100}
}
Response 200: {
  "delegation_id": "uuid",
  "output": "Replication lag is 2ms — healthy",
  "status": "success",
  "from_agent": "dba-specialist",
  "completed_at": "2026-03-13T..."
}
Response 404: {"error": "agent not found: <name>"}
Response 503: {"error": "agent inbox full"}
```

Handler logic:
1. Deserialize request body to `DelegateRequest { from_agent: String, task: String, payload: Value }`
2. Call `manager.delegate(name, &req.task, req.payload).await`
3. Serialize `DelegationResult` as response
4. On error: map `AgentixError::runtime` to 503, not-found to 404

Wire the route in the existing axum router.
