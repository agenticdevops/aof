```rust
use anyhow::Result;
use crate::api::CliContext;

pub async fn memory_list(cli: &CliContext, agent: &str) -> Result<()> {
    let url = format!("{}/api/v1/agents/{}/memory", cli.gateway_url, agent);
    let resp: serde_json::Value = cli.client.get(&url).send().await?.json().await?;

    if let Some(entries) = resp.as_array() {
        if entries.is_empty() {
            println!("No memory entries found for agent: {agent}");
            return Ok(());
        }
        println!("{:<38} {:<12} {:<50} {}", "ID", "RUN", "TEXT", "STORED_AT");
        println!("{}", "-".repeat(120));
        for entry in entries {
            let id = entry["id"].as_str().unwrap_or("-");
            let run_id = entry["run_id"].as_str().unwrap_or("-");
            let text = entry["text"].as_str().unwrap_or("-");
            let stored_at = entry["stored_at"].as_str().unwrap_or("-");
            let text_snippet = if text.len() > 48 { &text[..48] } else { text };
            println!("{id:<38} {run_id:<12} {text_snippet:<50} {stored_at}");
        }
    }
    Ok(())
}

pub async fn memory_clear(cli: &CliContext, agent: &str, yes: bool) -> Result<()> {
    if !yes {
        print!("Clear all memory for agent '{agent}'? This cannot be undone. [y/N] ");
        use std::io::{self, Write};
        io::stdout().flush()?;
        let mut input = String::new();
        io::stdin().read_line(&mut input)?;
        if !input.trim().eq_ignore_ascii_case("y") {
            println!("Cancelled.");
            return Ok(());
        }
    }
    let url = format!("{}/api/v1/agents/{}/memory", cli.gateway_url, agent);
    cli.client.delete(&url).send().await?;
    println!("Memory cleared for agent: {agent}");
    Ok(())
}
```

```
GET  /api/v1/agents/:name/memory
  Response: [{id, run_id, text, score, metadata, stored_at}]

DELETE /api/v1/agents/:name/memory
  Response: {"cleared": true, "agent": "<name>"}
```

GET handler:
1. Retrieve the agent's SqliteVectorBackend from AgentManager (manager must expose a method like `memory_backend_for(agent_name) -> Option<Arc<dyn VectorMemoryBackend>>`)
2. Return all entries — implement a `list_all(agent_id) -> AgentixResult<Vec<VectorEntry>>` method on SqliteVectorBackend, or use search_similar with an all-ones query vector as a workaround
3. Serialize and return as JSON array

Alternatively: add `list_entries(agent_id: &str) -> AgentixResult<Vec<VectorEntry>>` to `VectorMemoryBackend` trait and `SqliteVectorBackend`. This is the cleaner approach:

```sql
SELECT id, agent_id, run_id, text, metadata, stored_at
FROM vector_memory WHERE agent_id = ?1 ORDER BY stored_at DESC
```

DELETE handler:
1. Call `backend.clear_agent_memory(agent_name).await`
2. Return `{"cleared": true, "agent": name}`

```markdown
## [2.0.0-alpha.4] - 2026-03-13

### Added
- **Agent Coordination (Phase 16)**: Coordinator agents can now delegate tasks to specialist agents via `POST /api/v1/agents/:name/delegate` and receive structured results
- **Parallel delegation**: `CoordinatorProtocol::delegate_parallel()` fans out N tasks concurrently and returns results in order
- **Audit trail for delegations**: Every agent-to-agent delegation is recorded in the run audit log with parent/child context
- **Vector memory**: Agents with `spec.memory.enabled=true` store run contexts as embeddings and recall relevant past contexts before each run
- **SQLite vector backend**: Default vector memory backend stores embeddings as BLOBs in SQLite with cosine-similarity search in Rust
- **Per-agent memory isolation**: Agent memories are scoped by `agent_id` — agents cannot access each other's memories unless explicitly configured
- **Research phase**: Agents with `spec.research_phase.enabled=true` run a pre-loop fact-gathering step to reduce hallucinations
- **`agentix memory list <agent>`**: Lists all stored memory entries for an agent
- **`agentix memory clear <agent>`**: Clears all memory entries for an agent
- **REST memory API**: `GET/DELETE /api/v1/agents/:name/memory` for programmatic memory management
- **Quickstart examples**: `ops-coordinator.yaml` and `dba-specialist.yaml` demonstrate the coordination pattern

- CORE-04 (Agent memory recall)
```
