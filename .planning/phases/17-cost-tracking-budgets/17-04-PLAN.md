```rust
async fn list_all_costs(State(manager): State<Arc<AgentManager>>) -> impl IntoResponse {
    let summaries = manager.get_all_cost_summaries();
    // Sort by total_cost_usd descending
    let mut sorted = summaries;
    sorted.sort_by(|a, b| b.total_cost_usd.partial_cmp(&a.total_cost_usd).unwrap_or(std::cmp::Ordering::Equal));
    Json(sorted)
}

async fn get_agent_costs(
    State(manager): State<Arc<AgentManager>>,
    AxumPath(name): AxumPath<String>,
) -> impl IntoResponse {
    match manager.get_agent_cost_summary(&name) {
        Some(summary) => Json(summary).into_response(),
        None => (StatusCode::NOT_FOUND, Json(ErrorResponse { error: format!("No cost data for agent '{name}'"), field: None })).into_response(),
    }
}

async fn get_agent_run_costs(
    State(manager): State<Arc<AgentManager>>,
    AxumPath(name): AxumPath<String>,
    Query(params): Query<HashMap<String, String>>,
) -> impl IntoResponse {
    let limit = params.get("limit").and_then(|l| l.parse().ok()).unwrap_or(20usize);
    let runs = manager.get_agent_run_cost_summaries(&name, limit);
    Json(runs)
}
```

**Add to client.rs:**
```rust
pub async fn get_costs(&self) -> anyhow::Result<Vec<serde_json::Value>> {
    let url = format!("{}/api/v1/costs", self.gateway_url);
    let resp = self.client.get(&url).send().await?;
    Ok(resp.json().await?)
}

pub async fn get_agent_costs(&self, name: &str) -> anyhow::Result<serde_json::Value> {
    let url = format!("{}/api/v1/costs/agents/{}", self.gateway_url, name);
    let resp = self.client.get(&url).send().await?;
    Ok(resp.json().await?)
}

pub async fn get_agent_run_costs(&self, name: &str, limit: usize) -> anyhow::Result<Vec<serde_json::Value>> {
    let url = format!("{}/api/v1/costs/agents/{}/runs?limit={}", self.gateway_url, name, limit);
    let resp = self.client.get(&url).send().await?;
    Ok(resp.json().await?)
}
```

`costs_summary(ctx: &CliContext)`:
- Call `client.get_costs()`
- Print header: `"Cost Summary — All Agents"`

`costs_agent(ctx: &CliContext, name: &str, limit: usize)`:
- Call `client.get_agent_costs(name)` for the summary line
- Call `client.get_agent_run_costs(name, limit)` for the run table
- Print: `"Cost Breakdown — {name}"` then summary line (`Total: $X.XXXX across N runs`)
- Print per-run table: `Run ID (first 8 chars) | Model | Input | Output | Cost | Date (UTC)`
