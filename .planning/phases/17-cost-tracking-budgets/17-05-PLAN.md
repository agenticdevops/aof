Concrete implementation: In `AgentManager::new()`, change signature to accept an optional data directory path:
```rust
pub fn new(workspace: Option<WorkspaceConfig>) -> Self {
    // use ":memory:" for cost_store (tests)
    // This is overridden by with_db_path()
}

pub fn with_db_path(mut self, data_dir: &Path) -> Self {
    std::fs::create_dir_all(data_dir).ok();
    let cost_db = data_dir.join("cost.db");
    if let Ok(store) = CostStore::open(cost_db.to_str().unwrap_or(":memory:")) {
        self.cost_store = Arc::new(store);
    }
    self
}
```
