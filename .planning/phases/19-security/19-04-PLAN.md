**SSRF check before HTTP tool execution:**

Before calling `executor.execute(tool, input)`, if the tool is of type HTTP (or if the tool input contains a URL field) AND `self.config.ssrf_guard` is Some:
```rust
if let Some(ref guard) = self.config.ssrf_guard {
    // Extract URL from tool input if available
    if let Some(url) = input.get("url").and_then(|v| v.as_str()) {
        if let Err(violation) = guard.check_url(url) {
            // Log security violation to audit store
            if let Some(ref audit) = self.config.audit_store {
                let entry = AuditEntry {
                    id: None,
                    timestamp: Utc::now(),
                    event_type: AuditEventType::SecurityViolation,
                    agent_name: self.config.agent_name.clone(),
                    run_id: Some(run_id.to_string()),
                    actor: format!("agent:{}", self.config.agent_name),
                    action: format!("SSRF blocked: {}", violation),
                    outcome: AuditOutcome::Blocked(violation.to_string()),
                    details: {
                        let mut d = HashMap::new();
                        d.insert("tool_name".to_string(), json!(tool_name));
                        d.insert("blocked_url".to_string(), json!(url));
                        d
                    },
                    trace_id: trace_id.clone(),
                };
                let _ = audit.log_event(entry);
            }
            // Return SSRF error as tool result instead of executing
            return Err(format!("SSRF protection: {}", violation));
        }
    }
}
```

**Audit logging for tool calls:**

After each tool call completes (success or error), log an AuditEntry:
```rust
if let Some(ref audit) = self.config.audit_store {
    let entry = AuditEntry {
        id: None,
        timestamp: Utc::now(),
        event_type: AuditEventType::ToolCall,
        agent_name: self.config.agent_name.clone(),
        run_id: Some(run_id.to_string()),
        actor: format!("agent:{}", self.config.agent_name),
        action: format!("Tool call: {}", tool_name),
        outcome: match &tool_result {
            Ok(_) => AuditOutcome::Success,
            Err(e) => AuditOutcome::Failure(e.to_string()),
        },
        details: { /* tool_name, tool_type, duration_ms, input summary */ },

    };
    let _ = audit.log_event(entry);
}
```
