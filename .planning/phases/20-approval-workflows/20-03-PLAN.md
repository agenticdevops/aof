Add `approval_policy: Option<ApprovalPolicy>` and `approval_store: Option<Arc<ApprovalStore>>` fields to `ReActConfig`.

In the ReAct loop's tool execution section (just before calling `tool_executor.execute()`), add the approval gate logic:

```rust
// Before tool execution:
if let (Some(policy), Some(store)) = (&config.approval_policy, &config.approval_store) {
    if policy.requires_approval(&tool_call.name, &tool_call.input) {
        // 1. Create approval request
        let request = ApprovalRequest::new(
            &run_id, &agent_name,
            &format!("Tool call: {} with {}", tool_call.name, tool_call.input),
            Some(tool_call.name.clone()),
            Some(tool_call.input.clone()),
            policy.default_timeout_secs,
        );
        store.create_request(&request)?;

        // 2. Emit ApprovalRequested event
        let _ = event_tx.send(ReActEvent::ApprovalRequested { ... });

        // 3. Poll for decision
        loop {
            tokio::time::sleep(Duration::from_secs(1)).await;
            let updated = store.get_request(&request.id)?;
            match updated.status {
                ApprovalStatus::Approved { .. } => {
                    // emit ApprovalResolved, log audit, break to execute
                    break;
                }
                ApprovalStatus::Denied { reason, .. } => {
                    // emit ApprovalResolved, log audit, skip tool call
                    observation = format!("Action denied: {}", reason.unwrap_or_default());
                    continue 'iteration; // skip to next iteration
                }
                ApprovalStatus::TimedOut { .. } => {
                    observation = "Action timed out waiting for approval".to_string();
                    continue 'iteration;
                }
                ApprovalStatus::Pending => {
                    if request.is_expired() {
                        store.update_status(&request.id, &ApprovalStatus::TimedOut { expired_at: Utc::now() })?;
                        // handle timeout
                    }
                    continue; // keep polling
                }
            }
        }
    }
}
```
