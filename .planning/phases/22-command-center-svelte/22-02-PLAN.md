       ```rust
           AgentStatus { agent_name: String, status: String },
           RunStarted { agent_name: String, run_id: String },
           RunCompleted { agent_name: String, run_id: String, status: String, duration_ms: Option<u64> },
           ApprovalRequested { id: String, agent_name: String, action: String },
           ApprovalDecided { id: String, decision: String },

       }
       ```

       ```rust
       pub struct EventBroadcaster {

       impl EventBroadcaster {

       }
       ```

