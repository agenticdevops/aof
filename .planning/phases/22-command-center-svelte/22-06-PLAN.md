export interface ApprovalRequest {

  run_id: string;

POST   /api/v1/approvals/:id/approve  (body: { approver?: string })
POST   /api/v1/approvals/:id/deny     (body: { approver?: string, reason?: string })

<!-- WebSocket events -->
{ type: "approval_requested", id, agent_name, action }
{ type: "approval_decided", id, decision }

  <action>
    1. Create `src/lib/stores/approvals.ts`:

       **Stores:**
       - `approvals` writable store: `ApprovalRequest[]`
       - `approvalsLoading` writable store: `boolean`
       - `approvalsError` writable store: `string | null`

       **Functions:**
       - `loadApprovals()`: calls `api.approvals.list()`, updates store, handles errors
       - `approveRequest(id: string)`: calls `api.approvals.approve(id)`, then either optimistically update the local store (set status to 'approved') or reload. Return success/failure.
       - `denyRequest(id: string, reason?: string)`: calls `api.approvals.deny(id, undefined, reason)`, update local store similarly.
       - `pendingApprovals` derived store: filters approvals where status === 'pending'
       - `decidedApprovals` derived store: filters approvals where status !== 'pending'

       **WebSocket subscription:**
       - Subscribe to `wsEvents` store
       - On `approval_requested` event: add a new pending ApprovalRequest to the store (or reload from API if partial data)
       - On `approval_decided` event: update the matching request's status in-place

    2. Handle optimistic updates: when user clicks Approve/Deny, immediately update the UI (move from pending to decided), then confirm with API. If API fails, revert and show error toast.
  </action>
